only = set(sys.argv[1:])
summary = []
for ident, label, path, old, new, tests in MUTANTS:
    if only and ident not in only:
        continue
    file = ROOT / path
    original = file.read_text()
    if original.count(old) != 1:
        summary.append(f'{ident} NOT APPLIED (found {original.count(old)} times): {label}')
        continue
    file.write_text(original.replace(old, new))
    try:
        command = [str(ROOT / 'mvnw'), '-o', '-Dtest=' + tests, '-Dsurefire.failIfNoSpecifiedTests=false', '-Djacoco.skip=true', 'test']
        result = subprocess.run(command, cwd=ROOT, capture_output=True, text=True)
        log = OUT / f'{ident}.log'
        log.write_text(f'# {ident} {label}\n# file: {path}\n# - {old}\n# + {new}\n# command: {" ".join(command[1:])}\n# exit: {result.returncode}\n' + result.stdout + result.stderr)
        lines = [line for line in result.stdout.splitlines() if 'Tests run:' in line and ' in ' not in line]
        summary.append(f'{ident} exit={result.returncode} {lines[-1].strip() if lines else "(no tests line)"} :: {label}')
    finally:
        file.write_text(original)
    print(summary[-1], flush=True)
(OUT / 'summary.txt').write_text('\n'.join(summary) + '\n') if not only else None
print('\n'.join(summary))
