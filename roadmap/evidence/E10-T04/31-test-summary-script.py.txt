import pathlib, xml.etree.ElementTree as ET
# E10-T04 round 2: totals of target/surefire-reports and target/failsafe-reports, plus one line per class the round adds or changes.
CHANGED = ['SupportAccessIT', 'ConsoleProviderLockIT', 'CurrentUserFilterSupportTest', 'JndiDomainDnsTest', 'JobsApiIT', 'PlatformDomainsIT',
           'EventCatalogContractTest', 'AuditContractTest', 'AuditWriterTest', 'E2ContractIT', 'E10ContractIT', 'OpenApiSnapshotTest',
           'HealthControllerTest', 'ArchitectureTest', 'ArchitectureRulesTest']
for folder in ['target/surefire-reports', 'target/failsafe-reports']:
    totals = [0, 0, 0, 0]; rows = {}
    for report in sorted(pathlib.Path(folder).glob('TEST-*.xml')):
        suite = ET.parse(report).getroot()
        values = [int(suite.get(k, 0)) for k in ('tests', 'failures', 'errors', 'skipped')]
        totals = [a + b for a, b in zip(totals, values)]
        name = suite.get('name')
        if name.split('.')[-1] in CHANGED:
            rows[name] = values
    print(f'{folder}: tests={totals[0]}, failures={totals[1]}, errors={totals[2]}, skipped={totals[3]}')
    for name, v in sorted(rows.items()):
        print(f'  {name}: tests={v[0]}, failures={v[1]}, errors={v[2]}, skipped={v[3]}')
