SEED = "DemoPlatformAccountsSeedIT"
MUTANTS = [
    ("M01", "R-17-03: three failures make it BROKEN", MAIN + "platform/application/PlatformDomains.java",
     "public static final int BROKEN_AFTER = 3;", "public static final int BROKEN_AFTER = 4;", RECHECK, None),
    ("M02", "P9: only due domains (platform.domainRecheckDays)", MAIN + "platform/application/PlatformDomains.java",

    ("M05", "P9: an apex domain passes with its addresses", MAIN + "platform/application/PlatformDomains.java",

    ("M06", "console: a BROKEN apex domain re-verifies by ADDRESS", MAIN + "platform/application/PlatformDomains.java",

    ("M11", "E118: the demo account gets AGILITYHUB_ADMIN", MAIN + "identity/application/DemoPlatformAccountsSeeder.java",

summary = []
for mid, rule, path, old, new, it, unit in MUTANTS:
    file = ROOT / path
    original = file.read_text()
    assert original.count(old) == 1, f"{mid}: the original text is not unique in {path}"
    file.write_text(original.replace(old, new))
    try:
        command = ["./mvnw", "-o", "-q", "verify", "-Djacoco.skip=true", "-Dsurefire.failIfNoSpecifiedTests=false",
                   "-Dfailsafe.failIfNoSpecifiedTests=false", "-Dtest=" + (unit or "none"), "-Dit.test=" + (it or "none")]
        result = subprocess.run(command, cwd=ROOT, text=True, stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
    finally:
        file.write_text(original)
    (EVIDENCE / f"11-mutant-{mid}.log").write_text(result.stdout)
    tests = [line.strip() for line in result.stdout.splitlines() if "Tests run:" in line and ("FAIL" in line or "ERROR" in line)]
    line = f"{mid} exit={result.returncode} {tests[-1] if tests else '(no test summary: compile or setup failure)'} :: {rule}"
    print(line, flush=True)
    summary.append(line)
(EVIDENCE / "11-mutants-summary.log").write_text("\n".join(summary) + "\n")
