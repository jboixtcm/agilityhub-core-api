from pathlib import Path
import re

pages = sorted(Path('docs/playbooks').glob('*.md'))
assert {p.name for p in pages} == {
    'checklist.md', 'new-entity.md', 'new-endpoint.md',
    'new-scheduler.md', 'new-event-consumer.md',
}
pattern = re.compile(
    r'Source: \[(src/[^\]]+)\]\(../../\1#L(\d+)-L(\d+)\)\.\n\n'
    r'```java\n(.*?)\n```', re.S,
)
links = excerpts = 0
for page in pages:
    text = page.read_text()
    lines = len(text.splitlines())
    assert lines <= 150, (page, lines)
    assert '<!-- excerpt:' not in text
    matches = list(pattern.finditer(text))
    assert text.count('```java\n') == len(matches), page
    for match in matches:
        source, start, end, code = match.groups()
        source_lines = Path(source).read_text().splitlines()
        assert code == '\n'.join(source_lines[int(start)-1:int(end)]), source
        kind = 'test-classes' if source.startswith('src/test/') else 'classes'
        relative = source.split('/java/', 1)[1]
        compiled = Path('target') / kind / Path(relative).with_suffix('.class')
        assert compiled.is_file(), compiled
        print(f'EXCERPT {page.name}: {source}:{start}-{end} exact; class compiled')
    for target in re.findall(r'\[[^\]]+\]\(([^)]+)\)', text):
        assert (page.parent / target.split('#', 1)[0]).resolve().exists(), target
        links += 1
    excerpts += len(matches)
    print(f'PASS {page.name}: {lines} lines; {len(matches)} Java excerpts')
print(f'PASS: 5 playbooks; {excerpts} exact Java excerpts; {links} local links')
